3. Then the user task is used for task approval by specific user present part of loop iteration.
There is user form by using which user can assign or decline the output value will be assigned to "managerapproval".

4. The XOR gateway decides if managerapproval = "approved" then it will execute AfterApproval service task, otherwise the script task will be executed.


5. As part of the service task we are setting the "approvalResult" variable not the "approvalResults" as in the Outputelement in camunda modeler we have specified as "approvalResult".  
"approvalResults" will be calculated by camunda engine by default after each iteration is completed.

package com.cam.camunda.jobworker;

import java.util.HashMap;
import java.util.Map;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.stereotype.Component;
import org.springframework.util.StringUtils;

import io.camunda.client.annotation.JobWorker;
import io.camunda.client.api.response.ActivatedJob;
import io.camunda.client.api.worker.JobClient;

@Component
public class AfterApproval {

	Logger logger = LoggerFactory.getLogger(getClass());

	@JobWorker(type = "after_assign_approval")
	public void afterAssigneeApproval(ActivatedJob activatedJob, JobClient jobClient) {

		Map<String, Object> resultVarsMAp = activatedJob.getVariablesAsMap();

		logger.info("All Variables:"+resultVarsMAp.toString());
		if (resultVarsMAp != null && resultVarsMAp.containsKey("managerapproval")
				&& null != resultVarsMAp.get("managerapproval")) {
			String managerapproval = (String) resultVarsMAp.get("managerapproval");

			logger.info("After Loan approval result:" + managerapproval);

			String approver = (activatedJob.getVariable("approver") != null)
					? activatedJob.getVariable("approver").toString()
					: null;

			logger.info("after assigned approver:" + approver);

			Map<String, Object> approvalResult = new HashMap<String, Object>();
			approvalResult.put("approver", approver);
			approvalResult.put("decision", managerapproval);

			jobClient.newCompleteCommand(activatedJob.getKey()).variables(Map.of("CustomerOnBoardingStatus",
					managerapproval.equalsIgnoreCase("approved") ? "APPROVED" : "DECLINED", "approvalResult",approvalResult))
			.send()
			.join();
		} else {

			logger.info("Error Else - Loan approval result:");

			jobClient.newFailCommand(activatedJob).retries(activatedJob.getRetries() - 1).send().join();
		}
	}

}

6. This service task is not inside embeded subprocess  and started when after the multi instance activity completed. 
In service task we could able to access the "approvalResults" output variable generated by multiinstance having List of "approvalResult" [{approver=manager1, decision=approved}, {approver=manager2, decision=approved}, {approver=manager3, decision=approved}].
Multiinstance is completed based upon the completion condition "numberOfCompletedInstances =  3" SPECIFIED IN bpmn. This is a predefined variable present for camunda.

package com.cam.camunda.jobworker;

import java.util.List;
import java.util.Map;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.stereotype.Component;

import io.camunda.client.annotation.JobWorker;
import io.camunda.client.api.response.ActivatedJob;
import io.camunda.client.api.worker.JobClient;


@Component
public class CustomerOnBoardingApproval {

	Logger LOG = LoggerFactory.getLogger(getClass());
	
	@JobWorker(type = "validate_onboarding_request")
	public void onboarding(final ActivatedJob activatedJob, final JobClient jobClient) {
		
		Map<String, Object> resultVarsMAp= activatedJob.getVariablesAsMap();
		
		LOG.info("After All Approvals all variables:"+resultVarsMAp.toString());
		
		if(resultVarsMAp != null && resultVarsMAp.containsKey("approvalResults") && null != resultVarsMAp.get("approvalResults")) {
			List<String> approvalList = (List<String>) resultVarsMAp.get("approvalResults");
		
		LOG.info("After All Approvals approvalResults:"+approvalList);
		
		jobClient.newCompleteCommand(activatedJob.getKey())
         .send()
         .join();
		
		}else {
			
			LOG.info("Error Else - After All Approvals approvalResults");
			
			jobClient.newFailCommand(activatedJob)
			.retries(activatedJob.getRetries()-1)
			.send()
			.join();
		}
		
	}
	
}
